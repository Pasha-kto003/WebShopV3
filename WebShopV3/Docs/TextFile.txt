@section Scripts {
    <script>
    let selectedComponents = [];
    let totalPrice = 0;
    let componentsTotalPrice = 0;
    let markupAmount = 0;

    function calculatePrices() {
        // Сумма компонентов
        componentsTotalPrice = selectedComponents.reduce((sum, component) => sum + component.price, 0);

        // Наценка 10%
        markupAmount = componentsTotalPrice * 0.1;

        // Итоговая цена
        totalPrice = componentsTotalPrice + markupAmount;

        // Обновляем отображение
        updatePriceDisplay();
    }

    // Обновление отображения цен
    function updatePriceDisplay() {
        const componentsTotalElement = document.getElementById('componentsTotal');
        const markupAmountElement = document.getElementById('markupAmount');
        const totalPriceElement = document.getElementById('totalPrice');

        if (componentsTotalElement) {
            componentsTotalElement.textContent = formatPrice(componentsTotalPrice);
        }

        if (markupAmountElement) {
            markupAmountElement.textContent = formatPrice(markupAmount);
        }

        if (totalPriceElement) {
            totalPriceElement.textContent = formatPrice(totalPrice);
        }
    }

    // Инициализация при загрузке страницы
    // Переключение выбора компонента
    function toggleComponent(id, type, name, price, socket, memoryType, formFactor, powerConnector) {
        const component = {
            id: id,
            type: type,
            name: name,
            price: price,
            socket: socket,
            memoryType: memoryType,
            formFactor: formFactor,
            powerConnector: powerConnector
        };

        const existingIndex = selectedComponents.findIndex(sc => sc.id === id);
        
        if (existingIndex > -1) {
            // Удаляем компонент
            selectedComponents.splice(existingIndex, 1);
        } else {
            // Добавляем компонент, проверяя тип
            const existingSameType = selectedComponents.find(sc => sc.type === type);
            if (existingSameType) {
                if (!confirm(`Заменить ${getComponentTypeName(type)} "${existingSameType.name}" на "${name}"?`)) {
                    return;
                }
                // Удаляем старый компонент того же типа
                selectedComponents = selectedComponents.filter(sc => sc.type !== type);
            }
            selectedComponents.push(component);
        }

        updateConfigurationDisplay();
        updateComponentCards();
        checkCompatibility();
        calculatePrices();
        autoFillDescription();
    }

    function autoFillDescription() {
        const descriptionTextarea = document.getElementById('computerDescription');
        if (!descriptionTextarea) return;

        // Проверяем, не ввел ли пользователь свое описание вручную
        const currentDescription = descriptionTextarea.value.trim();

        // Если поле пустое или содержит только автогенерированный текст, обновляем
        if (!currentDescription || 
            currentDescription.startsWith('Собранный ПК:') ||
            currentDescription.startsWith('Конфигурация ПК включает:')) {
    
            // Генерируем новое описание
            const newDescription = generateAutoDescription();
            descriptionTextarea.value = newDescription;
    
            // Обновляем счетчик символов
            if (typeof updateCounter === 'function') {
                const descriptionCounter = document.getElementById('descriptionCounter');
                updateCounter(descriptionTextarea, descriptionCounter, 500);
            }
        }
    }

    function updateConfigurationDisplay() {
        const configurationList = document.getElementById('configurationList');
        const saveButton = document.getElementById('saveConfiguration');

        if (selectedComponents.length === 0) {
            configurationList.innerHTML = `

                    <div class="text-4xl mb-2">🖥️</div>

            if (saveButton) saveButton.disabled = true;

        }

                    </div>
                    <div class="text-right">

                    </div>

        `).join('');

        if (saveButton) saveButton.disabled = false;
    }

    // Получение деталей компонента для отображения
    function getComponentDetails(component) {
        const details = [];
        
        if (component.socket) details.push(`Сокет: ${component.socket}`);
        if (component.memoryType) details.push(`Память: ${component.memoryType}`);
        if (component.formFactor) details.push(`Форм-фактор: ${component.formFactor}`);
        if (component.powerConnector) details.push(`Разъемы: ${component.powerConnector}`);
        
        return details.length > 0 ? 
            `<div class="text-gray-500 text-xs mt-1">${details.join(', ')}</div>` : '';
    }

    // Обновление карточек компонентов (подсветка выбранных)
    function updateComponentCards() {
        document.querySelectorAll('.component-card').forEach(card => {
            const componentId = parseInt(card.getAttribute('data-component-id'));
            const isSelected = selectedComponents.some(sc => sc.id === componentId);
            
            card.classList.toggle('border-primary', isSelected);
            card.classList.toggle('border-gray-700', !isSelected);
            
            const button = card.querySelector('button');
            if (button) {
                button.textContent = isSelected ? 'Удалить' : 'Добавить';
                button.className = `text-xs ${isSelected ? 'text-red-400' : 'text-primary'} hover:underline`;
            }
        });
    }

    // Проверка совместимости компонентов
    async function checkCompatibility() {
        const compatibilityResult = document.getElementById('compatibilityResult');
        
        if (selectedComponents.length === 0) {
            if (compatibilityResult) compatibilityResult.innerHTML = '';
            return;
        }

        // Показываем загрузку
        if (compatibilityResult) {
            compatibilityResult.innerHTML = `
                <div class="text-gray-400 text-sm">
                    🔄 Проверка совместимости...
                </div>
            `;
        }

        try {
            const componentIds = selectedComponents.map(c => c.id);
            const response = await fetch('@Url.Action("CheckCompatibility", "PcBuilder")', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify(componentIds)
            });

            if (response.ok) {
                const result = await response.json();
                displayCompatibilityResult(result);
            } else {
                throw new Error('Ошибка проверки совместимости');
            }
        } catch (error) {
            if (compatibilityResult) {
                compatibilityResult.innerHTML = `
                    <div class="bg-red-500/20 border border-red-500/30 rounded-lg p-3">
                        <div class="text-red-400 text-sm">❌ Ошибка проверки совместимости</div>
                    </div>
                `;
            }
        }
    }

    // Отображение результатов проверки совместимости
    function displayCompatibilityResult(result) {
        const compatibilityResult = document.getElementById('compatibilityResult');
        if (!compatibilityResult) return;
        
        if (result.isCompatible) {
            compatibilityResult.innerHTML = `
                <div class="bg-green-500/20 border border-green-500/30 rounded-lg p-3">
                    <div class="text-green-400 text-sm font-semibold mb-1">✅ ${result.successMessage || 'Все компоненты совместимы!'}</div>
                    ${result.warnings && result.warnings.length > 0 ? `
                        <div class="text-yellow-400 text-xs mt-1">
                            ${result.warnings.join('<br>')}
                        </div>
                    ` : ''}
                </div>
            `;
        } else {
            compatibilityResult.innerHTML = `
                <div class="bg-red-500/20 border border-red-500/30 rounded-lg p-3">
                    <div class="text-red-400 text-sm font-semibold mb-2">❌ Обнаружены проблемы совместимости:</div>
                    <div class="text-red-300 text-xs space-y-1">
                        ${result.errors.map(error => `• ${error}`).join('<br>')}
                    </div>
                </div>
            `;
        }
    }

    // Вспомогательная функция для форматирования цены
    function formatPrice(price) {
        return new Intl.NumberFormat('ru-RU', {
            style: 'currency',
            currency: 'RUB',
            minimumFractionDigits: 0
        }).format(price);
    }

    function setupImageUpload() {
        const imageInput = document.getElementById('computerImage');
        if (!imageInput) return;
        
        const uploadArea = imageInput.closest('label');
        if (!uploadArea) return;

        // Добавляем обработчики для drag and drop
        uploadArea.addEventListener('dragover', function(e) {
            e.preventDefault();
            uploadArea.classList.add('border-primary', 'bg-primary/10');
        });

        uploadArea.addEventListener('dragleave', function(e) {
            e.preventDefault();
            uploadArea.classList.remove('border-primary', 'bg-primary/10');
        });

        uploadArea.addEventListener('drop', function(e) {
            e.preventDefault();
            uploadArea.classList.remove('border-primary', 'bg-primary/10');
        
            const files = e.dataTransfer.files;
            if (files.length > 0) {
                imageInput.files = files;
                handleImageSelection(files[0]);
            }
        });

        // Обработчик выбора файла через диалог
        imageInput.addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (file) {
                handleImageSelection(file);
            }
        });

        // Функция обработки выбранного изображения
        function handleImageSelection(file) {
            // Проверяем размер файла (макс. 5MB)
            if (file.size > 5 * 1024 * 1024) {
                alert('Файл слишком большой. Максимальный размер: 5MB');
                imageInput.value = '';
                return;
            }

            // Проверяем тип файла
            const validTypes = ['image/jpeg', 'image/jpg', 'image/png', 'image/webp'];
            if (!validTypes.includes(file.type)) {
                alert('Пожалуйста, выберите файл в формате JPG, PNG или WebP');
                imageInput.value = '';
                return;
            }

            // Показываем превью
            const reader = new FileReader();
            reader.onload = function(e) {
                let imagePreview = document.getElementById('imagePreview');
                let previewImage = document.getElementById('previewImage');
                let removeImageBtn = document.getElementById('removeImage');
                
                if (!imagePreview) {
                    imagePreview = document.createElement('div');
                    imagePreview.id = 'imagePreview';
                    imagePreview.className = 'mt-3';
                    imagePreview.innerHTML = `
                        <p class="text-sm text-gray-400 mb-2">Предпросмотр:</p>
                        <div class="relative inline-block">
                            <img id="previewImage" class="w-32 h-32 object-cover rounded-lg border border-gray-600" alt="Предпросмотр">
                            <button type="button" id="removeImage" class="absolute -top-2 -right-2 bg-red-500 text-white text-xs w-6 h-6 rounded-full hover:bg-red-600">
                                ✕
                            </button>
                        </div>
                    `;
                    imageInput.parentNode.appendChild(imagePreview);
                    
                    // Находим элементы после создания
                    previewImage = document.getElementById('previewImage');
                    removeImageBtn = document.getElementById('removeImage');
                    
                    // Добавляем обработчик удаления
                    removeImageBtn.addEventListener('click', function() {
                        imageInput.value = '';
                        imagePreview.remove();
                    });
                }
                
                previewImage.src = e.target.result;
                imagePreview.classList.remove('hidden');
                
                // Скрываем текущее изображение при загрузке нового
                const currentImage = document.getElementById('currentImage');
                if (currentImage) {
                    currentImage.classList.add('hidden');
                }
            };
            reader.onerror = function() {
                alert('Ошибка при чтении файла');
                imageInput.value = '';
            };
            reader.readAsDataURL(file);
        }
    }

    // Счетчики символов для полей ввода
    function setupInputCounters() {
        const nameInput = document.getElementById('computerName');
        const descriptionInput = document.getElementById('computerDescription');
        const nameCounter = document.getElementById('nameCounter');
        const descriptionCounter = document.getElementById('descriptionCounter');

        if (!nameInput || !descriptionInput || !nameCounter || !descriptionCounter) return;

        function updateCounter(input, counter, maxLength) {
            const count = input.value.length;
            counter.textContent = `${count}/${maxLength}`;
    
            if (count > maxLength * 0.8) {
                counter.className = 'text-xs text-yellow-400 mt-1';
            } else {
                counter.className = 'text-xs text-gray-400 mt-1';
            }
        }

        nameInput.addEventListener('input', () => {
            updateCounter(nameInput, nameCounter, 100);
        });

        descriptionInput.addEventListener('input', () => {
            updateCounter(descriptionInput, descriptionCounter, 500);
        });

        // Инициализируем счетчики
        updateCounter(nameInput, nameCounter, 100);
        updateCounter(descriptionInput, descriptionCounter, 500);
    }

    // Сохранение конфигурации
    async function saveConfiguration() {
        const computerName = document.getElementById('computerName')?.value.trim();
        const computerDescription = document.getElementById('computerDescription')?.value.trim();
        const imageInput = document.getElementById('computerImage');
        const imageFile = imageInput?.files[0];

        // Валидация названия
        if (!computerName) {
            alert('Пожалуйста, введите название конфигурации');
            document.getElementById('computerName')?.focus();
            return;
        }

        if (computerName.length > 100) {
            alert('Название не должно превышать 100 символов');
            document.getElementById('computerName')?.focus();
            return;
        }

        if (computerDescription && computerDescription.length > 500) {
            alert('Описание не должно превышать 500 символов');
            document.getElementById('computerDescription')?.focus();
            return;
        }

        if (!computerDescription && selectedComponents.length > 0) {
            if (confirm('Хотите автоматически сгенерировать описание на основе выбранных компонентов?')) {
                const autoDescription = generateAutoDescription();
                document.getElementById('computerDescription').value = autoDescription;
            }
        }

        // Проверяем, что выбраны обязательные компоненты
        const requiredTypes = ['MB', 'CPU', 'RAM', 'PSU', 'Case'];
        const missingTypes = requiredTypes.filter(type => 
            !selectedComponents.some(sc => sc.type === type)
        );

        if (missingTypes.length > 0) {
            const typeNames = {
                'MB': 'Материнская плата',
                'CPU': 'Процессор', 
                'RAM': 'Оперативная память',
                'PSU': 'Блок питания',
                'Case': 'Корпус'
            };
    
            const missingNames = missingTypes.map(type => typeNames[type]).join(', ');
            alert(`Пожалуйста, выберите обязательные компоненты: ${missingNames}`);
            return;
        }

        // Валидация изображения
        if (imageFile) {
            // Проверяем размер файла (макс. 5MB)
            if (imageFile.size > 5 * 1024 * 1024) {
                alert('Файл слишком большой. Максимальный размер: 5MB');
                return;
            }

            // Проверяем тип файла
            const validTypes = ['image/jpeg', 'image/jpg', 'image/png', 'image/webp'];
            if (!validTypes.includes(imageFile.type)) {
                alert('Пожалуйста, выберите файл в формате JPG, PNG или WebP');
                return;
            }
        }

        // Показываем индикатор загрузки
        const saveButton = document.getElementById('saveConfiguration');
        if (!saveButton) return;
        
        const originalText = saveButton.innerHTML;
        saveButton.disabled = true;
        saveButton.innerHTML = `
            <div class="inline-block animate-spin rounded-full h-5 w-5 border-b-2 border-dark mr-2"></div>
            Сохранение...
        `;

        try {
            // Создаем FormData для отправки файла
            const formData = new FormData();
            formData.append('Name', computerName);
            formData.append('Description', computerDescription || generateAutoDescription());
            formData.append('Price', totalPrice.toString());
            formData.append('Quantity', '1');
    
            // Добавляем компоненты
            selectedComponents.forEach((component) => {
                formData.append('ComponentIds', component.id.toString());
            });

            // Добавляем ID компьютера если редактируем
            const computerId = @(ViewBag.ComputerId ?? "null");
            if (computerId) {
                formData.append('ComputerId', computerId.toString());
            }

            // Добавляем файл изображения если выбран
            if (imageFile) {
                formData.append('ImageFile', imageFile);
            }

            const token = document.querySelector('input[name="__RequestVerificationToken"]')?.value;
    
            if (!token) {
                throw new Error('Токен безопасности не найден');
            }

            const response = await fetch('@Url.Action("SaveConfiguration", "PcBuilder")', {
                method: 'POST',
                headers: {
                    'RequestVerificationToken': token
                },
                body: formData
            });

            if (response.ok) {
                const result = await response.json();
                if (result.success) {
                    alert(result.message);
                    window.location.href = '@Url.Action("Details", "Computer")' + `/${result.computerId}`;
                } else {
                    alert('Ошибка при сохранении конфигурации: ' + (result.message || 'Неизвестная ошибка'));
                }
            } else {
                throw new Error('Ошибка сервера при сохранении');
            }
        } catch (error) {
            console.error('Ошибка при сохранении конфигурации:', error);
            alert('Ошибка при сохранении конфигурации: ' + error.message);
        } finally {
            // Восстанавливаем кнопку
            if (saveButton) {
                saveButton.disabled = false;
                saveButton.innerHTML = originalText;
            }
        }
    }

    function getComponentTypeName(type) {
        const typeNames = {
            'MB': 'Материнская плата',
            'CPU': 'Процессор',
            'RAM': 'Оперативная память',
            'GPU': 'Видеокарта',
            'PSU': 'Блок питания',
            'Case': 'Корпус',
            'SSD': 'SSD накопитель',
            'Cooler': 'Охлаждение'
        };
        return typeNames[type] || type;
    }

    function generateAutoDescription() {
        if (selectedComponents.length === 0) {
            return "Выберите компоненты для сборки";
        }
        
        // Сортируем компоненты в правильном порядке
        const orderedComponents = [...selectedComponents].sort((a, b) => {
            const order = {
                'MB': 1,      // Материнская плата
                'CPU': 2,     // Процессор
                'RAM': 3,     // Оперативная память
                'GPU': 4,     // Видеокарта
                'PSU': 5,     // Блок питания
                'Case': 6,    // Корпус
                'SSD': 7,     // SSD
                'HDD': 8,     // HDD
                'Cooler': 9   // Охлаждение
            };
            
            return (order[a.type] || 10) - (order[b.type] || 10);
        });
        
        // Собираем названия компонентов
        const componentNames = orderedComponents.map(c => c.name);
        
        // Формируем описание
        return `Собранный ПК: ${componentNames.join(', ')}`;
    }

    function generateDetailedAutoDescription() {
        if (selectedComponents.length === 0) {
            return "Выберите компоненты для сборки";
        }

        const componentDescriptions = selectedComponents.map(component => {
            const typeName = getComponentTypeName(component.type);
            return `${typeName}: ${component.name}`;
        });

        return `Конфигурация ПК включает:\n${componentDescriptions.join('\n')}`;
    }

    function generateAndFillDescription() {
        const descriptionTextarea = document.getElementById('computerDescription');
        if (!descriptionTextarea) return;

        if (selectedComponents.length === 0) {
            alert('Выберите хотя бы один компонент для генерации описания');
            return;
        }

        // Генерируем описание
        const newDescription = generateAutoDescription();
        descriptionTextarea.value = newDescription;

        // Обновляем счетчик символов
        const descriptionCounter = document.getElementById('descriptionCounter');
        if (descriptionCounter && typeof updateCounter === 'function') {
            updateCounter(descriptionTextarea, descriptionCounter, 500);
        }

        // Показываем уведомление
        showNotification('Описание сгенерировано автоматически', 'info');
    }

    // Функция для очистки описания
    function clearDescription() {
        const descriptionTextarea = document.getElementById('computerDescription');
        if (!descriptionTextarea) return;

        if (descriptionTextarea.value.trim() && !confirm('Очистить описание?')) {
            return;
        }

        descriptionTextarea.value = '';

        // Обновляем счетчик символов
        const descriptionCounter = document.getElementById('descriptionCounter');
        if (descriptionCounter && typeof updateCounter === 'function') {
            updateCounter(descriptionTextarea, descriptionCounter, 500);
        }
    }

    function showNotification(message, type = 'info') {
        const notification = document.createElement('div');
        notification.className = `fixed top-4 right-4 z-50 px-4 py-3 rounded-lg shadow-lg border transform transition-all duration-300 ${
            type === 'info' 
                ? 'bg-blue-900/90 border-blue-700 text-blue-100' 
                : 'bg-green-900/90 border-green-700 text-green-100'
        }`;
        notification.innerHTML = `
            <div class="flex items-center gap-2">
                <svg class="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                    <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M13 16h-1v-4h-1m1-4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
                </svg>
                <span class="text-sm">${message}</span>
            </div>
        `;

        document.body.appendChild(notification);

        // Автоматическое скрытие через 3 секунды
        setTimeout(() => {
            notification.style.opacity = '0';
            notification.style.transform = 'translateX(100%)';
            setTimeout(() => notification.remove(), 300);
        }, 3000);
    }
